cmake_minimum_required(VERSION 3.20)
project(JoggleExtensionOracle LANGUAGES C CXX)
find_package(MLIR REQUIRED CONFIG)
if(NOT EXISTS "${EXTENSION_SOURCE}")
  message(FATAL_ERROR "EXTENSION_SOURCE must name the candidate C++ implementation")
endif()
add_executable(extension-oracle mlir-driver.cpp "${EXTENSION_SOURCE}")
if(EXTENSION_COMPOUND)
  if(EXTENSION_INPUT_FORMAT OR EXTENSION_REWRITE OR EXTENSION_DEFINITION OR EXTENSION_CONSTRUCT)
    message(FATAL_ERROR "compound mode cannot be combined with another task mode")
  endif()
  target_compile_definitions(extension-oracle PRIVATE EXTENSION_COMPOUND)
endif()
if(EXTENSION_INPUT_FORMAT)
  if(EXTENSION_REWRITE OR EXTENSION_DEFINITION OR EXTENSION_CONSTRUCT)
    message(FATAL_ERROR "input format mode cannot be combined with another task mode")
  endif()
  target_compile_definitions(extension-oracle PRIVATE EXTENSION_INPUT_FORMAT)
endif()
if(EXTENSION_REWRITE)
  target_compile_definitions(extension-oracle PRIVATE EXTENSION_REWRITE)
endif()
if(EXTENSION_DEFINITION)
  target_compile_definitions(extension-oracle PRIVATE EXTENSION_DEFINITION)
endif()
if(EXTENSION_CONSTRUCT)
  if(NOT EXTENSION_DEFINITION)
    message(FATAL_ERROR "EXTENSION_CONSTRUCT requires EXTENSION_DEFINITION")
  endif()
  target_compile_definitions(extension-oracle PRIVATE EXTENSION_CONSTRUCT)
endif()
target_compile_features(extension-oracle PRIVATE cxx_std_17)
target_include_directories(extension-oracle SYSTEM PRIVATE
  ${MLIR_INCLUDE_DIRS} ${LLVM_INCLUDE_DIRS})
separate_arguments(EXTENSION_LLVM_DEFINITIONS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(extension-oracle PRIVATE ${EXTENSION_LLVM_DEFINITIONS})
target_link_libraries(extension-oracle PRIVATE MLIRParser MLIRIR MLIRSupport MLIRFuncDialect)
